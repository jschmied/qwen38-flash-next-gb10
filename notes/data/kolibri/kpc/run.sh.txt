#!/usr/bin/env bash
# Kolibri 1 on tensorfold-native (kolibri1-forward + slot prefix reuse): agent-loop TTFT, reuse on vs off.
D=/home/jschmied/kolibri-drafter/kpc; B=/home/jschmied/tfbuild-kz/native/bin/tensorfold-native
M=/opt/llm/models/kolibri-1; P=8099
for arm in on_a off_a on_b off_b; do
  r=1; [[ $arm == off* ]] && r=0
  env TENSORFOLD_KOLIBRI_KERNELS=/home/jschmied/kolibri-drafter/kset-sm121 TENSORFOLD_KOLIBRI_STREAMS=2 TENSORFOLD_KOLIBRI_REUSE=$r $B serve $M --host 127.0.0.1 --port $P --name m --context 32768 --parallel 1 --no-update-check --no-thinking --prompt-cache-gib 0 > $D/serve-$arm.log 2>&1 &
  pid=$!
  for i in $(seq 1 120); do curl -s -m 2 http://127.0.0.1:$P/v1/models | grep -q '"m"' && break; kill -0 $pid 2>/dev/null || break; sleep 5; done
  echo "== $arm $(grep -m1 -o 'loaded in [0-9.]*s' $D/serve-$arm.log)"
  python3 /home/jschmied/kolibri-drafter/pcache/agentloop.py http://127.0.0.1:$P $arm 12 >> $D/results.jsonl 2>> $D/probe-err.log
  kill $pid; wait $pid 2>/dev/null
  sleep 10
done
echo "== KPC DONE"
