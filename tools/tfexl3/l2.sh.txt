#!/bin/bash
R=/opt/llm/runners/results/tfexl3; L=/opt/llm/runners/tfexl3; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw
export HOME=/root PYTHONUNBUFFERED=1
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
cd /home/jschmied/git/tf-exl3-grp && TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-grp PYTHONPATH=$PWD/src timeout 2400 \
  $PY -m pytest -q -p no:cacheprovider tests/cuda/test_exl3_experts.py tests/cuda/test_qwen4_exp_exl3.py tests/cuda/test_group_kernel_smem.py > $R/l2-tests.txt 2>&1
echo "tests rc=$?" >> $R/l2-tests.txt
for rep in a b; do
  memok; TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-061 $PY $L/prefill_ab.py /home/jschmied/git/tf-exl3-061/src $M pr184-$rep > $R/l2-pr184-$rep.txt 2>&1
  memok; TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-grp $PY $L/prefill_ab.py /home/jschmied/git/tf-exl3-grp/src $M grp-$rep > $R/l2-grp-$rep.txt 2>&1
done
for rep in a b c; do
  TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-061 $PY $L/small_bench.py /home/jschmied/git/tf-exl3-061/src pr184-$rep >> $R/l2-small.txt 2>&1
  TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-grp $PY $L/small_bench.py /home/jschmied/git/tf-exl3-grp/src grp-$rep >> $R/l2-small.txt 2>&1
done
echo "== L2 DONE ==" >> $R/l2-tests.txt
