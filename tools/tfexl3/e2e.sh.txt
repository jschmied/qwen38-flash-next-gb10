#!/bin/bash
# T13 round 6: EXL3 prefill 8k/32k, stock vs tile-work-list branch, alternating, two rounds.
R=/opt/llm/runners/results/tfexl3; L=/opt/llm/runners/tfexl3; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw
export HOME=/root PYTHONUNBUFFERED=1
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
for rep in a b; do
  memok; TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext $PY $L/prefill_ab.py /home/jschmied/git/tensorfold/src $M stock-$rep > $R/e2e-stock-$rep.txt 2>&1
  memok; TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3exp $PY $L/prefill_ab.py /home/jschmied/git/tf-exl3-prefill/src $M branch-$rep > $R/e2e-branch-$rep.txt 2>&1
  echo "round $rep done" >> $R/e2e-chain.txt
done
echo "== E2E DONE ==" >> $R/e2e-chain.txt
