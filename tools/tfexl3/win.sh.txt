#!/bin/bash
# Round 8: EXL3 prefill, #184 vs #184 + MOE_WINDOW 2048, alternating, two rounds; then the qwen4_exp EXL3 GPU test.
R=/opt/llm/runners/results/tfexl3; L=/opt/llm/runners/tfexl3; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw
export HOME=/root PYTHONUNBUFFERED=1 TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-061
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
for rep in a b; do
  memok; $PY $L/prefill_ab.py /home/jschmied/git/tf-exl3-061/src $M pr184-$rep > $R/win-pr184-$rep.txt 2>&1
  memok; $PY $L/prefill_ab.py /home/jschmied/git/tf-exl3-win/src $M win2048-$rep > $R/win-2048-$rep.txt 2>&1
  echo "round $rep done" >> $R/win-chain.txt
done
cd /home/jschmied/git/tf-exl3-win && PYTHONPATH=$PWD/src timeout 1800 $PY -m pytest -q -p no:cacheprovider tests/cuda/test_qwen4_exp_exl3.py > $R/win-tests.txt 2>&1
echo "tests rc=$?" >> $R/win-chain.txt
echo "== WIN DONE ==" >> $R/win-chain.txt
