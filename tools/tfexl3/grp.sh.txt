#!/bin/bash
# Round 9: parallel grouping — EXL3 GPU tests, routed() hashes/timing vs #184 (2 rounds), kernel profile.
R=/opt/llm/runners/results/tfexl3; L=/opt/llm/runners/tfexl3; PY=/opt/llm/runtime/tf-venv/bin/python

  TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-exl3-grp $PY $L/routed_bench.py /home/jschmied/git/tf-exl3-grp/src > $R/grp-branch-$rep.txt 2>&1
done
