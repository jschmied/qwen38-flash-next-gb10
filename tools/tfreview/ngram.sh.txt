#!/bin/bash
# #252 cold-table prefill A/B; starts after the #250 chain.
R=/opt/llm/runners/results/tfreview; L=/opt/llm/runners/tfreview; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw; SRC=/home/jschmied/git/tf-pr252/src
export HOME=/root PYTHONUNBUFFERED=1 TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-pr252
until grep -q "COST DONE" $R/cost-chain.txt 2>/dev/null; do sleep 30; done
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
for rep in a b; do
  for arm in "off|0|0" "ahead|1|0" "rand|1|1"; do
    IFS='|' read name ah rnd <<< "$arm"; memok
    env TF_NGRAM_AHEAD=$ah NGRAM_RANDOM=$rnd timeout 2400 $PY $L/ngram_cold.py $SRC $M $name-$rep >> $R/ngram.jsonl 2> $R/ngram-$name-$rep.err
    echo "$(date +%T) $name-$rep rc=$?" >> $R/ngram-chain.txt
  done
done
echo "== NGRAM DONE ==" >> $R/ngram-chain.txt
