#!/usr/bin/env bash
# Run 16j saves 2,500-32,500 (scored at 35k pause, superseded by 35,000+): to PBS, sha256-verified, then dropped.
set -uo pipefail
D=/home/jschmied/kolibri-drafter; DEST=/mnt/bulk/gb10/kolibri-drafter
K="-o BatchMode=yes -i /home/jschmied/.ssh/id_ed25519 -o UserKnownHostsFile=/home/jschmied/.ssh/known_hosts"
SSH="ssh -n $K root@10.0.0.70"
PROTECT=""
DROP=" out16j/drafter-step2500.pt out16j/drafter-step5000.pt out16j/drafter-step7500.pt out16j/drafter-step10000.pt out16j/drafter-step12500.pt out16j/drafter-step15000.pt out16j/drafter-step17500.pt out16j/drafter-step20000.pt out16j/drafter-step22500.pt out16j/drafter-step25000.pt out16j/drafter-step27500.pt out16j/drafter-step30000.pt out16j/drafter-step32500.pt"
log(){ echo "[$(date +%H:%M:%S)] $*"; }
cd $D
for p in $PROTECT $DROP; do [ -e "$p" ] || continue
  rsync -aR --partial -e "ssh $K" "./$p" root@10.0.0.70:$DEST/ || { log "rsync FAILED $p"; exit 1; }
done
log "synced"
verify(){  # $1 path -> prints mismatch count
  local l r; l=$(cd $D && find "$1" -type f -print0 | sort -z | xargs -0 sha256sum)
  r=$($SSH "cd $DEST && find '$1' -type f -print0 | sort -z | xargs -0 sha256sum")
  comm -23 <(echo "$l" | sort) <(echo "$r" | sort) | wc -l
}
for p in $PROTECT; do [ -e "$p" ] || continue
  n=$(verify "$p"); log "protect $p: $(find $p -type f | wc -l) files, $n not matched on PBS"; done
for p in $DROP; do [ -e "$p" ] || continue
  n=$(verify "$p")
  if [ "$n" = 0 ]; then rm -rf "$D/$p"; log "dropped $p (verified)"; else log "KEPT $p: $n files not matched"; fi
done
log "$(df -h --output=avail /home | tail -1) free"
log "== BACKUPCLEAN5 DONE =="
