#!/usr/bin/env bash
# Root, unit fx-kolibri-drafter-run12 (watchdog "window"). Waits for run 11's scoring, then run 12 = run 11 final +
# glm3 (70M GLM tokens, shards 10-17 + 30), recipe C; scored with run 11 final and run 10 on the same clean recordings
# (after 2026-10-05 09:55: neither run 10's successors nor these runs trained on any recording). Server stays down.
set -uo pipefail
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src; V=$D/draft_vocab_32k.json
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
until grep -q "== RUN11 DONE ==" $D/run11.log 2>/dev/null; do sleep 120; done
systemctl is-active -q fx-kolibri-serve && { log "ABORT: server is up (someone restored serving)"; exit 1; }
cp -p $D/out11/drafter.pt $D/drafter-run11-final.pt
log "spec bench (run 11 final in the engine, GPU idle)"
U $PY $T/specbench.py /opt/llm/models/kolibri-1 $D/drafter-run11-final.pt --vocab $V --swe $D/bench \
  --chat $D/own_chat.jsonl --de $D/own_chat_de.jsonl --depths 1,2,3 --tokens 256 --n 8 --concurrent 4 \
  > $D/specbench-run11.log 2>&1 || log "specbench rc=$? (continuing)"
log "spec bench done: $(grep -c tok_s $D/specbench-run11.log) cells"
systemctl stop fx-kolibri-keepckpt
systemd-run --unit=fx-kolibri-keepckpt --collect --uid=jschmied bash -c "bash $D/keepckpt.sh $D/out12 >> $D/keepckpt.log 2>&1"
log "run 12 training"
U $PY $T/train_mt.py /opt/llm/models/kolibri-1 $D/swe $D/out12 --extra $D/glm/glm3 --swe-passes 0 --hold 3 \
  --init $D/drafter-run11-final.pt --taps 44,47,49 --rollout 3 --lr 2e-4 --eval 2000 --plateau -1 --loss kll1 --l1w 0.1 \
  --draft-vocab $V --log 50 --save 500 > $D/train12.log 2>&1
log "train rc=$?"
sleep 90
CK="$D/drafter-run10-window.pt $D/drafter-run11-final.pt $(ls $D/out12/drafter-step*.pt | sort -V | awk 'NR%20==0' | tr '\n' ' ') $D/out12/drafter.pt"
E(){ U bash -c "ulimit -n 1048576 2>/dev/null || ulimit -n \$(ulimit -Hn); $PY $T/eval_rec.py /opt/llm/models/kolibri-1 --data $D/rec/live --after '2026-10-05 09:55' $1 --draft-vocab $V --ckpt $CK" 2>&1 | grep -v Warning; }
log "fresh SWE";    E "--contains 'returncode|interact with a computer shell' --excludes django"
log "fresh chat";   E "--excludes 'returncode|interact with a computer shell| und | nicht | ist '"
log "fresh German"; E "--contains ' und | nicht | ist ' --excludes 'returncode|interact with a computer shell'"
log "== RUN12 DONE =="
