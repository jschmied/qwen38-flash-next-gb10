#!/usr/bin/env bash
# Root, unit fx-kolibri-drafter-run15 (watchdog "window"). Run 15 = block drafter whose position 0 IS run 11's chain
# drafter (model_block chain_ctx, from_chain): context rows fc(cat(embed(next token), fuse(taps))), block row 0 = the
# anchor's row, rows 1-3 mask + the anchor's fused state, causal within the block; fuse/fc/layer 0/out_norm copied from
# run 11 final (FFN padded 4096 -> 6144 with zero-output columns), layers 1-3 start as identity. Untrained it scores
# fresh chat accepted_3 0.667 (position 1 0.637 = run 11's step 1), above 14e's 0.54 after 23M tokens.
# Data: glm + glm2 + glm3 + de4k x3 from the start, recordings before 2026-10-05 09:55 at 30 % of steps (SWE 2 passes,
# chat 3), 1,024 anchors, on-policy weights, assistant anchors only; position decay 0.8 fixed; lr 4e-4 cosine to 4e-5.
# Hypothesis: positions 2-4 rise fast on top of a fixed position 1; chat accepted_3 passes run 11's 1.23 by ~20M tokens.
# Out of range: position 1 falling below 0.60 (training disturbs the copied layer) -> lower lr.
set -uo pipefail
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src; V=$D/draft_vocab_32k.json
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
systemctl stop fx-kolibri-keepckpt
mkdir -p $D/out15; chown jschmied: $D/out15
systemd-run --unit=fx-kolibri-keepckpt --collect --uid=jschmied bash -c "bash $D/keepckpt.sh $D/out15 >> $D/keepckpt.log 2>&1"
log "run 15 block training from run 11"
U $PY $T/train_block.py /opt/llm/models/kolibri-1 $D/out15 --data $D/glm/glm $D/glm/glm2 $D/glm/glm3 $D/glm/de4k \
  $D/glm/de4k $D/glm/de4k --draft-vocab $V --log 50 --save 2500 --from-chain $D/drafter-run11-final.pt \
  --rec $D/rec/live --rec-share 0.3 --rec-passes-swe 2 --rec-passes 3 --anchors 1024 --onpolicy --assistant-only \
  --decay 0.8 --lr 4e-4 --lr-end 4e-5 --warmup 300 > $D/train15.log 2>&1
log "train rc=$?"
log "== RUN15 DONE =="
