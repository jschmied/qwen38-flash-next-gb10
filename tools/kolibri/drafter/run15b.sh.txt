#!/usr/bin/env bash
# Root, unit fx-kolibri-drafter-run15b (watchdog "window"). Run 15b: position 0 IS run 11's chain drafter, frozen
# (row0_chain: its output after layer 0; fuse, fc, layer 0, out_norm reloaded from run 11 final and frozen); layers 1-3
# and the mask train for positions 1-3 only. Starts from loss-A/B arm A (KL, tied with accept-until-fail and acc-CE).
# Why: in run 15 training wore position 1 down from 0.637 to 0.570 (chat fresh) while positions 2-4 learned.
# Init check: chat position 1 0.6366 = run 11 exactly; positions 2-4 0.079/0.069/0.049 (they re-adapt to the restored
# layer 0). Data from arm A's point (4173199 tokens), recordings 30 % (passes halved: SWE 1, chat 2, windows replay).
# Hypothesis: positions 2-4 recover A's 0.25/0.15/0.10 within ~1,500 steps and keep rising on a fixed position 1; chat
# accepted_3 above A's 0.81 by step 2,500. Out of range: positions 2-4 still below A's at step 2,500.
set -uo pipefail
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src; V=$D/draft_vocab_32k.json
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
systemctl stop fx-kolibri-keepckpt
mkdir -p $D/out15b; chown jschmied: $D/out15b
systemd-run --unit=fx-kolibri-keepckpt --collect --uid=jschmied bash -c "bash $D/keepckpt.sh $D/out15b >> $D/keepckpt.log 2>&1"
log "run 15b: frozen chain position 0"
U $PY $T/train_block.py /opt/llm/models/kolibri-1 $D/out15b --data $D/glm/glm $D/glm/glm2 $D/glm/glm3 $D/glm/de4k \
  $D/glm/de4k $D/glm/de4k --draft-vocab $V --log 50 --save 2500 --init $D/out15ab-A/drafter.pt \
  --from-chain $D/drafter-run11-final.pt --row0-chain --skip-tokens 4173199 \
  --rec $D/rec/live --rec-share 0.3 --rec-passes-swe 1 --rec-passes 2 --anchors 1024 --onpolicy --assistant-only \
  --decay 0.8 --lr 4e-4 --lr-end 4e-5 --warmup 300 > $D/train15b.log 2>&1
log "train rc=$?"
log "== RUN15B DONE =="
