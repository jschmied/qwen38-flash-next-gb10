#!/usr/bin/env bash
# Continue the Kolibri drafter from the SWE checkpoint on fresh chat text plus one SWE pass; stop on held-out plateau.
set -uo pipefail
T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
D=/home/jschmied/kolibri-drafter
export PYTHONPATH=/home/jschmied/git/tf-kolibri/src
/opt/llm/runtime/tf-venv/bin/python $T/train.py /opt/llm/models/kolibri-1 $D/swe $D/out2 --extra $D/chat --swe-passes 1 \
  --init $D/drafter-swe-3500.pt --hold 3 --lr 2e-4 --eval 250 --log 50 > $D/train2.log 2>&1
echo "train rc=$?"
/opt/llm/runtime/tf-venv/bin/python $T/accept.py /opt/llm/models/kolibri-1 $D/swe $D/out2/drafter.pt --names $D/out2/held_out.json > $D/accept2.log 2>&1
echo "accept rc=$? $(tail -1 $D/accept2.log)"
