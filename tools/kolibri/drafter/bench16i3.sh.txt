#!/usr/bin/env bash
# Root. At run 16i's step 25,000 save: specbench the block drafter at depth 2, eager vs CUDA-graphed (fixed-shape pass,
# in-place cache rows), on the untouched prompts; then resume 16i.
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
until [ -e $D/out16i/drafter-step25000.pt ]; do sleep 30; done
sleep 30
log "16i step 25000 kept; stopping 16i"
systemctl stop fx-kolibri-drafter-run16i; while systemctl is-active -q fx-kolibri-drafter-run16i; do sleep 5; done
B="--swe $D/bench --chat $D/bench2/bench_chat.jsonl --de $D/bench2/bench_de.jsonl --tokens 256 --n 8 --concurrent 4"
for G in 0 1; do
  log "block drafter 16i 25000, depth 2, graph=$G"
  U env TENSORFOLD_KOLIBRI_BLOCK_GRAPH=$G timeout 3600 $PY $T/specbench.py /opt/llm/models/kolibri-1 $D/out16i/drafter-step25000.pt \
    --vocab $D/draft_vocab_64k.json --depths 2 $B > $D/specbench-16i-graph$G.log 2>&1; log "rc=$?"
done
systemd-run --unit=fx-kolibri-drafter-run16i --collect -E HOME=/root bash -c "bash $D/run16i.sh >> $D/run16i.log 2>&1"
log "16i resumed"
log "== BENCH16I3 DONE =="
