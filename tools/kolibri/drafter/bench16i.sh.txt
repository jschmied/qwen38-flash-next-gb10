#!/usr/bin/env bash
# Root. At run 16i's step 20,000 save: stop it, specbench the block drafter (16i 20,000, 64k vocab, depths 2 and 4) and
# run 11's chain (32k, depth 1) on the same untouched prompts (bench2: ultrachat 1,400+, alpaca-de 7,900+; SWE turns
# from bench/), greedy and served sampling, replies must match the copy-only arm; then resume 16i.
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
until [ -e $D/out16i/drafter-step20000.pt ]; do sleep 30; done
sleep 30
log "16i step 20000 kept; stopping 16i"
systemctl stop fx-kolibri-drafter-run16i; while systemctl is-active -q fx-kolibri-drafter-run16i; do sleep 5; done
B="--swe $D/bench --chat $D/bench2/bench_chat.jsonl --de $D/bench2/bench_de.jsonl --tokens 256 --n 8 --concurrent 4"
log "block drafter 16i 20000"
U timeout 5400 $PY $T/specbench.py /opt/llm/models/kolibri-1 $D/out16i/drafter-step20000.pt --vocab $D/draft_vocab_64k.json \
  --depths 2,4 $B > $D/specbench-16i.log 2>&1; log "rc=$?"
log "run 11 chain"
U timeout 5400 $PY $T/specbench.py /opt/llm/models/kolibri-1 $D/drafter-run11-final.pt --vocab $D/draft_vocab_32k.json \
  --depths 1 $B > $D/specbench-run11-bench2.log 2>&1; log "rc=$?"
systemd-run --unit=fx-kolibri-drafter-run16i --collect -E HOME=/root bash -c "bash $D/run16i.sh >> $D/run16i.log 2>&1"
log "16i resumed"
log "== BENCH16I DONE =="
