#!/usr/bin/env bash
# Root, unit fx-kolibri-drafter-run15c (watchdog "window"). Run 15c = run 15b continued by a FULL resume (weights,
# AdamW, step, data position, rng) from its step 17,500, writing on into out15b, with the recordings back at 10 % of
# steps for one more pass (--rec-refresh, SWE 1 / chat 1). Why: with the recordings used up SWE fell 1.217 -> 1.176
# (steps 5,000 -> 17,500): the GLM text has no agent sessions. Hypothesis: SWE holds or recovers at the next
# checkpoints while chat/German keep their +0.02 per 5,000 steps.
set -uo pipefail
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src; V=$D/draft_vocab_32k.json
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
cp $D/out15b/resume.pt $D/out15b/resume-step17500.pt
log "run 15c: resume 15b at 17,500 with recordings at 10 %"
U $PY $T/train_block.py /opt/llm/models/kolibri-1 $D/out15b --data $D/glm/glm $D/glm/glm2 $D/glm/glm3 $D/glm/de4k \
  $D/glm/de4k $D/glm/de4k --draft-vocab $V --log 50 --save 2500 --init $D/out15b/drafter-step17500.pt \
  --resume $D/out15b/resume-step17500.pt --rec-refresh \
  --rec $D/rec/live --rec-share 0.1 --rec-passes-swe 1 --rec-passes 1 --anchors 1024 --onpolicy --assistant-only \
  --decay 0.8 --lr 4e-4 --lr-end 4e-5 --warmup 300 > $D/train15c.log 2>&1
log "train rc=$?"
log "== RUN15C DONE =="
