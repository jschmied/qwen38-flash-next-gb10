#!/usr/bin/env bash
# Root. At run 16g's step 12,500 save: stop it, score arm U and 16g's 5,000 / 10,000 / 12,500 on the fixed slices,
# resume 16g from its resume.pt (run16g.sh resumes when out16g/resume.pt exists).
D=/home/jschmied/kolibri-drafter; T=/home/jschmied/git/qwen38-flash-next-gb10/tools/kolibri/drafter
PY=/opt/llm/runtime/tf-venv/bin/python; SRC=/home/jschmied/git/tf-kolibri/src; V=$D/draft_vocab_32k.json
log(){ echo "[$(date +%H:%M:%S)] $*"; }
U(){ runuser -u jschmied -- env PYTHONPATH=$SRC HOME=/home/jschmied "$@"; }
until [ -e $D/out16g/drafter-step12500.pt ]; do sleep 30; done
sleep 30
log "16g step 12500 kept, resume.pt $(stat -c %y $D/out16g/resume.pt | cut -c12-19); stopping 16g"
systemctl stop fx-kolibri-drafter-run16g; while systemctl is-active -q fx-kolibri-drafter-run16g; do sleep 5; done
CK="$D/out16e-U/drafter.pt $D/out16g/drafter-step5000.pt $D/out16g/drafter-step10000.pt $D/out16g/drafter-step12500.pt"
E(){ U bash -c "ulimit -n \$(ulimit -Hn); $PY $T/eval_block.py /opt/llm/models/kolibri-1 --data $D/rec/live --after '2026-10-05 09:55' $1 --draft-vocab $V --ckpt $CK" 2>&1 | grep '"ckpt"\|Error'; }
log "fresh SWE";    E "--contains 'returncode|interact with a computer shell' --excludes django"
log "fresh chat";   E "--excludes 'returncode|interact with a computer shell| und | nicht | ist '"
log "fresh German"; E "--contains ' und | nicht | ist ' --excludes 'returncode|interact with a computer shell'"
systemd-run --unit=fx-kolibri-drafter-run16g --collect -E HOME=/root bash -c "bash $D/run16g.sh >> $D/run16g.log 2>&1"
log "16g resumed"
log "== SCORE16G DONE =="
