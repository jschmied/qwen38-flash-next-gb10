# user 2026-09-30: "re run failed with bigger context". argv: <src arm> <rerun arm>. Writes /root/set-ovf-<rerun>-{py,js}.txt
# = the instances of <src arm> whose trajectory ended in a context overflow: vLLM's ContextWindowExceededError, or
# TensorFold's BadRequestError "... exceeding the server's 65536-token safe cache capacity" (v2, 2026-10-01).
SRC=$1; DST=$2; O=/root/fn-swe/out
: > /root/set-ovf-$DST-py.txt; : > /root/set-ovf-$DST-js.txt
