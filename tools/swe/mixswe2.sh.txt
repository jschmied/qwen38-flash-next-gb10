#!/bin/bash
# Replaces mixswe.sh (stopped 17:31 while its first probe, vllmhard1 hard, kept running). User 17:3x: "do both scenes once
# more" -> after that probe: scenes with seed 2 on vLLM then TensorFold, then the rest of the hard/mixed slices.
R=/opt/llm/runners/results; S=/tmp/claude-1000/-home-jschmied-git-dgx-spark-setup-guide/24a6e32a-d571-4261-96e4-1463dea3f47a/scratchpad; O=/opt/llm/runners/results/fishscene; G=/home/jschmied/git/qwen38-flash-next-gb10/tools/fishscene
T=/opt/llm/runtime/tf-venv; M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw; P=/opt/llm/runtime/vllm-venv-fnext/bin/python
log(){ echo "$(date +%T) $*" >> $R/mixswe.txt; }
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
ready(){ for i in $(seq 1 120); do sleep 10; systemctl is-active -q $1 || { log "$1 DIED"; return 1; }
  curl -sf -H "Authorization: Bearer none" http://127.0.0.1:8092/v1/models > /dev/null && { log "$1 ready after $((i*10)) s"; return 0; }; done; return 1; }
probe(){ su -s /bin/bash jschmied -c "bash $S/sweprobe2.sh $1 $2 > $S/swe-$1-$2.json 2> $S/swe-$1-$2.err"; log "$1 $2 rc=$? $(cat $S/swe-$1-$2.json)"; }
stopu(){ systemctl stop $1; log "$1 stopped"; sleep 20; memok; }
while kill -0 3954334 2>/dev/null; do sleep 30; done
log "vllmhard1 hard (from mixswe.sh) $(cat $S/swe-vllmhard1-hard.json)"
stopu fx-vllm-swe
# scenes, seed 2
systemd-run --unit=fx-vllm-fish --collect --uid=llm \
  -E FN_VENV=/opt/llm/runtime/vllm-venv-main1ea7 -E FN_MODEL=/opt/llm/models/qwen38-flash-next-mtpfp4 \
  -E FN_MAXLEN=131072 -E FN_SEQS=16 -E FN_PLE_OFFLOAD=0 \
  -E FN_SPEC_METHOD=mtp -E FN_SPEC_N=5 -E FN_SPEC_NODROP=1 -E FN_SPEC_LOCALARGMAX=0 \
  -E FN_DRAFT_VOCAB=/opt/llm/runners/dv/draft_vocab_32768.txt -E FN_DRAFT_HEAD_NVFP4=1 -E FN_NVFP4_CFG=64,4 \
  -E FN_DRAFT_PROB=1 -E FN_SPEC_DRAFTPROB=1 -E "FN_CG_SIZES=[1,2,4,6,8,12,16,18,24,30,36,48,60,72,96]" \
  -E FN_HCFUSE=1 -E FN_CG_MODE=FULL_AND_PIECEWISE -E FN_MOEFUSE=1 -E FN_GDNNQ=1 \
  -E "FN_EXTRA=--engram-config {\"checkpoint_mapped\":true} --use-replayssm --block-size 1728 --prefix-match-unit 64 --mamba-ssm-cache-dtype bfloat16 --kv-cache-memory-bytes 8589934592" \
  /opt/llm/serve-flashnext.sh
if ready fx-vllm-fish; then GEN_SEED=2 timeout 7200 $P $G/gen.py $O/vllm-prod-2 >> $R/mixswe.txt 2>&1; log "fish vllm seed2 rc=$?"; fi
stopu fx-vllm-fish
systemd-run --unit=fx-tf-fish --collect -E HOME=/root -E TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext -E PATH=/usr/local/cuda/bin:/usr/local/bin:/usr/bin:/bin \
  bash -c "$T/bin/tensorfold serve $M --port 8092 --name flashnext --context 131072 --max-tokens 100000 --parallel 1 > /opt/llm/tf-serve-fish2.log 2>&1"
if ready fx-tf-fish; then GEN_SEED=2 timeout 7200 $P $G/gen.py $O/tensorfold-exl3-2 >> $R/mixswe.txt 2>&1; log "fish tf seed2 rc=$?"; fi
stopu fx-tf-fish
python3 $G/check.py $O/tensorfold-exl3-2.html $O/vllm-prod-2.html >> $R/mixswe.txt 2>&1; chmod -R a+r $O; log "FISH2 DONE"
# rest of the vLLM slices
systemd-run --unit=fx-vllm-swe --collect --uid=llm \
  -E FN_VENV=/opt/llm/runtime/vllm-venv-main1ea7 -E FN_MODEL=/opt/llm/models/qwen38-flash-next-mtpfp4 \
  -E FN_MAXLEN=65536 -E FN_SEQS=16 -E FN_PLE_OFFLOAD=0 \
  -E FN_SPEC_METHOD=mtp -E FN_SPEC_N=5 -E FN_SPEC_NODROP=1 -E FN_SPEC_LOCALARGMAX=0 \
  -E FN_DRAFT_VOCAB=/opt/llm/runners/dv/draft_vocab_32768.txt -E FN_DRAFT_HEAD_NVFP4=1 -E FN_NVFP4_CFG=64,4 \
  -E FN_DRAFT_PROB=1 -E FN_SPEC_DRAFTPROB=1 -E "FN_CG_SIZES=[1,2,4,6,8,12,16,18,24,30,36,48,60,72,96]" \
  -E FN_HCFUSE=1 -E FN_CG_MODE=FULL_AND_PIECEWISE -E FN_MOEFUSE=1 -E FN_GDNNQ=1 \
  -E "FN_EXTRA=--engram-config {\"checkpoint_mapped\":true} --use-replayssm --block-size 1728 --prefix-match-unit 64 --mamba-ssm-cache-dtype bfloat16 --kv-cache-memory-bytes 8589934592" \
  /opt/llm/serve-flashnext.sh
if ready fx-vllm-swe; then probe vllmhard1 jshard; probe vllmmix2 pym; probe vllmmix2 jsm; fi
stopu fx-vllm-swe
systemd-run --unit=fx-tf-exl3 --collect -E HOME=/root -E TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext \
  -E PATH=/usr/local/cuda/bin:/usr/local/bin:/usr/bin:/bin \
  bash -c "$T/bin/tensorfold serve $M --port 8092 --name flashnext --context 65536 --max-tokens 16000 --parallel 4 > /opt/llm/tf-serve-exl3-mix.log 2>&1"
if ready fx-tf-exl3; then for r in 1 2; do probe tfhard$r hard; probe tfhard$r jshard; done; fi
systemctl stop fx-tf-exl3; log "tensorfold stopped"; echo "== MIXSWE DONE ==" >> $R/mixswe.txt
