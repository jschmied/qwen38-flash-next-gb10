#!/bin/bash
# Hard slice (user: "replace 10 easy with 5 harder and 5 hardest"): 10 Verified instances (3 ">4 hours" = all that exist,
# 7 "1-4 hours", seed 1), two runs per engine, after vLLM run 2. vLLM first (same config as vllmswe3), then TensorFold+EXL3
# (same flags as tfexl3r2). Mixed slice = the 20 kept Python instances (existing runs) + these 10.
R=/opt/llm/runners/results; S=/tmp/claude-1000/-home-jschmied-git-dgx-spark-setup-guide/24a6e32a-d571-4261-96e4-1463dea3f47a/scratchpad
T=/opt/llm/runtime/tf-venv; M=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw
log(){ echo "$(date +%T) $*" >> $R/hardswe.txt; }
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
ready(){ for i in $(seq 1 120); do sleep 10; systemctl is-active -q $1 || { log "$1 DIED"; return 1; }
  curl -sf -H "Authorization: Bearer none" http://127.0.0.1:8092/v1/models > /dev/null && { log "$1 ready after $((i*10)) s"; return 0; }; done; return 1; }
while systemctl is-active -q vllmswe-chain3 || systemctl is-active -q fish-chain2 || systemctl is-active -q tfexl3r2-chain; do sleep 30; done
for u in fx-tf-exl3 fx-tf-fish fx-vllm-fish fx-vllm-swe; do systemctl is-active -q $u && { systemctl stop $u; log "$u stopped"; }; done
memok
systemd-run --unit=fx-vllm-swe --collect --uid=llm \
  -E FN_VENV=/opt/llm/runtime/vllm-venv-main1ea7 -E FN_MODEL=/opt/llm/models/qwen38-flash-next-mtpfp4 \
  -E FN_MAXLEN=65536 -E FN_SEQS=16 -E FN_PLE_OFFLOAD=0 \
  -E FN_SPEC_METHOD=mtp -E FN_SPEC_N=5 -E FN_SPEC_NODROP=1 -E FN_SPEC_LOCALARGMAX=0 \
  -E FN_DRAFT_VOCAB=/opt/llm/runners/dv/draft_vocab_32768.txt -E FN_DRAFT_HEAD_NVFP4=1 -E FN_NVFP4_CFG=64,4 \
  -E FN_DRAFT_PROB=1 -E FN_SPEC_DRAFTPROB=1 -E "FN_CG_SIZES=[1,2,4,6,8,12,16,18,24,30,36,48,60,72,96]" \
  -E FN_HCFUSE=1 -E FN_CG_MODE=FULL_AND_PIECEWISE -E FN_MOEFUSE=1 -E FN_GDNNQ=1 \
  -E "FN_EXTRA=--engram-config {\"checkpoint_mapped\":true} --use-replayssm --block-size 1728 --prefix-match-unit 64 --mamba-ssm-cache-dtype bfloat16 --kv-cache-memory-bytes 8589934592" \
  /opt/llm/serve-flashnext.sh
if ready fx-vllm-swe; then for r in 1 2; do
  su -s /bin/bash jschmied -c "bash $S/sweprobe2.sh vllmhard$r hard > $S/swe-vllmhard$r.json 2> $S/swe-vllmhard$r.err"; log "vllmhard$r rc=$? $(cat $S/swe-vllmhard$r.json)"; done; fi
systemctl stop fx-vllm-swe; log "vllm stopped"; sleep 20; memok
systemd-run --unit=fx-tf-exl3 --collect -E HOME=/root -E TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext \
  -E PATH=/usr/local/cuda/bin:/usr/local/bin:/usr/bin:/bin \
  bash -c "$T/bin/tensorfold serve $M --port 8092 --name flashnext --context 65536 --max-tokens 16000 --parallel 4 > /opt/llm/tf-serve-exl3-hard.log 2>&1"
if ready fx-tf-exl3; then for r in 1 2; do
  su -s /bin/bash jschmied -c "bash $S/sweprobe2.sh tfhard$r hard > $S/swe-tfhard$r.json 2> $S/swe-tfhard$r.err"; log "tfhard$r rc=$? $(cat $S/swe-tfhard$r.json)"; done; fi
systemctl stop fx-tf-exl3; log "tensorfold stopped"; echo "== HARDSWE DONE ==" >> $R/hardswe.txt
