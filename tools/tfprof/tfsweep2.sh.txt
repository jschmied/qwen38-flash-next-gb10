#!/bin/bash
# Round 2: T9 v2 small mix; EXL3 MOE_WINDOW 512/1024/2048; our checkpoint chunk rows 2048/4096/8192. Hypotheses: tools/tfprof/HYPOTHESIS.md.
R=/opt/llm/runners/results/tfsweep2; mkdir -p $R; T=/opt/llm/runtime/tf-venv; P=/opt/llm/runtime/vllm-venv-fnext/bin/python
TF=/home/jschmied/git/tensorfold; EXP=/home/jschmied/git/tensorfold-exp/src; BR=/home/jschmied/git/tensorfold-ple/src
X3=/opt/llm/models/qwen38-flash-next-exl3-3.05bpw; FN=/opt/llm/models/qwen38-flash-next-mtpfp4; F=/opt/llm/runners/results/tf060/prefill_fast.json
log(){ echo "$(date +%T) $*" >> $R/chain.txt; }
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
serve(){ # $1 tag $2 model $3 PYTHONPATH $4 ext dir $5 extra env $6 args
  memok; systemd-run --unit=fx-tfsweep2 --collect -E HOME=/root -E TORCH_EXTENSIONS_DIR=$4 \
    -E PATH=/usr/local/cuda/bin:/usr/local/bin:/usr/bin:/bin -E PYTHONPATH=$3 $5 \
    bash -c "$T/bin/tensorfold serve $2 --port 8092 --name flashnext $6 > /opt/llm/tfsweep2-$1.log 2>&1"
  for i in $(seq 1 180); do sleep 5; systemctl is-active -q fx-tfsweep2 || { log "$1 DIED: $(grep -E 'Error|error' /opt/llm/tfsweep2-$1.log | tail -2 | tr '\n' ' ' | cut -c1-300)"; return 1; }
    curl -sf http://127.0.0.1:8092/v1/models >/dev/null && { log "$1 ready ($(grep -o 'startup estimate [^;]*' /opt/llm/tfsweep2-$1.log | head -1))"; return 0; }; done; log "$1 not ready"; return 1; }
greedy(){ $P - "$1" <<'PY' >> $R/greedy.txt
import json, sys, urllib.request, hashlib
items = json.load(open("/opt/llm/runners/results/tf060/prefill_fast.json"))["items"]
msg = [x for x in items if x["length"] == 8192][0]["messages"]
b = {"model": "flashnext", "messages": msg, "max_tokens": 64, "temperature": 0, "chat_template_kwargs": {"enable_thinking": False}}
d = json.loads(urllib.request.urlopen(urllib.request.Request("http://127.0.0.1:8092/v1/chat/completions", json.dumps(b).encode(), {"Content-Type": "application/json"}), timeout=600).read())
t = d["choices"][0]["message"]["content"] or ""
print(sys.argv[1], hashlib.sha256(t.encode()).hexdigest()[:16], json.dumps(t[:80]))
PY
}
stopu(){ systemctl stop fx-tfsweep2; sleep 15; }
# 1. T9 v2
if serve t9v2 $X3 $BR /opt/llm/.cache-tf/ext "" "--context 65536 --parallel 8"; then
  timeout 1800 $P /opt/llm/runners/tf060/longmix.py t9v2 $X3 1 32000 4 2000 10 > $R/t9v2.json 2> $R/t9v2.err; log "t9v2 rc=$?"
  $T/bin/python $TF/tools/bench_concurrent.py http://127.0.0.1:8092 flashnext --levels 1,2 --alone --serial > $R/t9v2-exact.txt 2>&1; log "t9v2 exact rc=$?"
fi; stopu
# 2. EXL3 MOE_WINDOW
for w in 1024 512 2048; do
  if serve win$w $X3 $EXP /opt/llm/.cache-tf/ext-exp "-E TF_MOE_WINDOW=$w" "--context 65536 --parallel 1"; then
    $T/bin/python $TF/tools/prefill_cold.py run http://127.0.0.1:8092 flashnext $F $R/win$w.json > $R/win$w.txt 2>&1; greedy win$w; log "win$w done"
  fi; stopu
done
# 3. our checkpoint chunk rows
for r in 2048 4096 8192; do
  if serve fnrows$r $FN $EXP /opt/llm/.cache-tf/ext-exp "-E TF_PREFILL_ROWS=$r" "--context 65536 --parallel 1"; then
    $T/bin/python $TF/tools/prefill_cold.py run http://127.0.0.1:8092 flashnext $F $R/fnrows$r.json > $R/fnrows$r.txt 2>&1; greedy fnrows$r; log "fnrows$r done"
  fi; stopu
done
echo "== TFSWEEP2 DONE ==" >> $R/chain.txt
