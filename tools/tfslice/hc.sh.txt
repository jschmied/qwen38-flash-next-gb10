#!/bin/bash
R=/opt/llm/runners/results/tfslice; L=/opt/llm/runners/tfreview; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-mtpfp4; G=/home/jschmied/git
export HOME=/root PYTHONUNBUFFERED=1 MAX_JOBS=4 TENSORFOLD_PREFILL_ROWS=8192
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
for rep in a b; do for arm in base:tf-nextbase hc:tf-next; do
  n=${arm%%:*}; memok
  TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-hc$n timeout 3600 $PY $L/prefill_ab260.py $G/${arm#*:}/src $M h$n-$rep >> $R/hc.txt 2> $R/hc-$n-$rep.err
  echo "$(date +%T) $n-$rep rc=$? $(grep "\"h$n-$rep\"" $R/hc.txt | grep -o '"tokens": [0-9]*\|"prefill_s": [0-9.]*\|"hash": "[0-9a-f]*"' | tr '\n' ' ') $(grep -h -o 'HC fusion self-check[^"]*' $R/hc.txt | tail -1)" >> $R/hc-chain.txt
done; done
echo "== HC DONE ==" >> $R/hc-chain.txt
