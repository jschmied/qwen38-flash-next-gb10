#!/bin/bash
R=/opt/llm/runners/results/tfslice; L=/opt/llm/runners/tfreview; PY=/opt/llm/runtime/tf-venv/bin/python
M=/opt/llm/models/qwen38-flash-next-mtpfp4; SRC=/home/jschmied/git/tf-slices/src
export HOME=/root PYTHONUNBUFFERED=1 MAX_JOBS=4 TORCH_EXTENSIONS_DIR=/opt/llm/.cache-tf/ext-slices
memok(){ until [ $(awk '/MemAvailable/{print int($2/1048576)}' /proc/meminfo) -ge 100 ]; do sleep 10; done; }
for rep in a b; do for rows in 2048 4096 8192; do
  memok
  TENSORFOLD_PREFILL_ROWS=$rows timeout 3600 $PY $L/prefill_ab260.py $SRC $M r$rows-$rep >> $R/rows.txt 2> $R/rows-$rows-$rep.err
  echo "$(date +%T) rows $rows-$rep rc=$? $(grep "\"r$rows-$rep\"" $R/rows.txt | grep -o '"tokens": [0-9]*\|"prefill_s": [0-9.]*\|"hash": "[0-9a-f]*"' | tr '\n' ' ') $(grep -h -m1 -i 'error\|Traceback' $R/rows-$rows-$rep.err)" >> $R/rows-chain.txt
done; done
echo "== ROWS DONE ==" >> $R/rows-chain.txt
